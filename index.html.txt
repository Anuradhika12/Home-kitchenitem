function renderResults(q){
  q=q.trim().toLowerCase();
  var pool=ARTICLES.slice();
  /* include categories as searchable entries */
  CATEGORIES.forEach(function(c){pool.push({slug:"kitchen",title:c.name,category:"Category",
    excerpt:c.desc,tag:"Category",searchOnly:true})});
  var hits=pool.filter(function(a){
    if(!q)return true;
    return (a.title+" "+a.category+" "+a.excerpt+" "+(a.tag||"")).toLowerCase().indexOf(q)>-1;
  });
  if(!hits.length){results.innerHTML='<p class="small muted" style="padding:.6rem .4rem">No matches. Try "kettle", "vacuum", or "storage".</p>';return}
  results.innerHTML=hits.slice(0,10).map(function(a){
    var href=a.searchOnly?"#/kitchen":"#/article/"+a.slug;
    return '<a class="search-result" href="'+href+'"><span class="badge">'+(a.tag||a.category)+'</span><strong>'+a.title+'</strong><small>'+a.excerpt+'</small></a>';
  }).join("");
  results.querySelectorAll("a").forEach(function(a){a.addEventListener("click",closeSearch)});
}
